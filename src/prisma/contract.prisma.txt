//use prisma-8
enum TournamentStatus {
  ACTIVE
  PAUSED
  ENDED
}

enum TournamentFormat {
  SINGLE_ELIMINATION
  GROUP_STAGE
}

enum TeamStatus {
  PENDING
  APPROVED
  REJECTED
}

enum MatchStatus {
  PENDING
  ONGOING
  COMPLETED
}

enum MatchType {
  GROUP_MATCH
  KNOCKOUT_MATCH
  FINAL
}

model Tournament {
  id              String                     @id @default(cuid(2))
  guildId         String
  name            String
  game            String

  format          TournamentFormat

  rules           String?
  rulesURL        String?

  registrationEnd TimestamptzString?

  status          TournamentStatus           @default(ACTIVE)

  createdAt       TimestamptzString          @default(now())
  updatedAt       temporal.updatedAtString()

  teams           Team[]
  matches         Match[]
  groups          Group[]

  @@map("tournament")
}

model Group {
  id           String                     @id @default(cuid(2))
  name         String
  tournamentId String
  tournament   Tournament                 @relation(fields: [tournamentId], references: [id], onDelete: Cascade)

  createdAt    TimestamptzString          @default(now())
  updatedAt    temporal.updatedAtString()

  teams        Team[]
  matches      Match[]

  @@map("group")
}

model Team {
  id             String                     @id @default(cuid(2))
  name           String
  tag            String?
  tournamentId   String
  tournament     Tournament                 @relation(fields: [tournamentId], references: [id], onDelete: Cascade)
  status         TeamStatus                 @default(PENDING)

  groupId        String?
  group          Group?                     @relation(fields: [groupId], references: [id])
  groupPoints    Int                        @default(0)

  createdAt      TimestamptzString          @default(now())
  updatedAt      temporal.updatedAtString()

  players        Player[]

  matchesAsTeamA Match[]                    @relation("MatchTeamA")
  matchesAsTeamB Match[]                    @relation("MatchTeamB")
  roundsWon      Round[]                    @relation("RoundWinner")

  @@unique([name, tournamentId])
  @@map("team")
}

model Match {
  id           String                     @id @default(cuid(2))
  tournamentId String
  tournament   Tournament                 @relation(fields: [tournamentId], references: [id], onDelete: Cascade)

  type         MatchType
  roundName    String
  status       MatchStatus                @default(PENDING)
  startTime    TimestamptzString?

  groupId      String?
  group        Group?                     @relation(fields: [groupId], references: [id])

  teamAId      String?
  teamA        Team?                      @relation("MatchTeamA", fields: [teamAId], references: [id])
  teamAScore   Int?

  teamBId      String
  teamB        Team                       @relation("MatchTeamB", fields: [teamBId], references: [id])
  teamBScore   Int?

  winnerId     String? // winnerId = null => draw

  bestOf       Int                        @default(3)
  rounds       Round[]

  nextMatchId  String?
  nextMatch    Match?                     @relation("BracketTree", fields: [nextMatchId], references: [id])
  prevMatches  Match[]                    @relation("BracketTree")

  createdAt    TimestamptzString          @default(now())
  updatedAt    temporal.updatedAtString()

  @@map("match")
}

model Round {
  id          String                     @id @default(cuid(2))
  matchId     String
  match       Match                      @relation(fields: [matchId], references: [id], onDelete: Cascade)

  roundIndex  Int

  winnerId    String?
  winner      Team?                      @relation("RoundWinner", fields: [winnerId], references: [id], onDelete: Cascade)

  teamAPoint  Int
  teamBPoint  Int

  screenshoot String
  duration    Int?

  mvpId       String?
  mvp         Player?                    @relation(fields: [mvpId], references: [id], onDelete: SetNull)

  createdAt   TimestamptzString          @default(now())
  updatedAt   temporal.updatedAtString()

  @@map("round")
  // matchMVPId  String?
}

model Player {
  id        String                     @id @default(cuid(2))
  ign       String
  discordId String?
  createdAt TimestamptzString          @default(now())
  updatedAt temporal.updatedAtString()

  teamId    String
  team      Team                       @relation(fields: [teamId], references: [id], onDelete: Cascade)

  roundsMvp Round[]

  @@index([discordId])
  @@map("player")
}
